        async function runSandboxSimulation() {

            const method = document.getElementById('sb-method').value;
            const path = document.getElementById('sb-path').value;
            const query_string = document.getElementById('sb-query').value;
            const body = document.getElementById('sb-body').value;

            resultBox.innerHTML = `
Evaluating AST Tokens...
`;

            try {
                const res = await fetch('/api/simulate', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ method, path, query_string, body, headers: { 'user-agent': 'Bastion-Sandbox-Tester/1.0' } })
                });
                const data = await res.json();

                let verdictBadge = data.blocked ? ' 403 Hard Block' : (data.action === 'CHALLENGE' ? ' CAPTCHA Challenge' : ' 200 Allowed');

                resultBox.innerHTML = `
                    

                        

                            FINAL DECISION
                            ${verdictBadge}

Rule Anomaly Weight (w1): ${data.rule_score || 0}

                        
Semantic Cosine Score (w2): ${data.semantic_score || 0}

                        
Matched Semantic Archetype: ${data.semantic_category || 'None'}

                    


                    

                        SEMANTIC AST & VECTOR ANALYSIS
                        

                            ${data.reason || 'Input conforms to valid application structure with no threat signatures detected.'}

${data.raw_payload || 'N/A'}

        // Live Threat Map Canvas Animation
        function initThreatMap() {
            const canvas = document.getElementById('threatMapCanvas');
            if (!canvas) return;
            const ctx = canvas.getContext('2d');

            function resize() {
                canvas.width = canvas.parentElement.clientWidth;
                canvas.height = canvas.parentElement.clientHeight;
            }
            resize();
            window.addEventListener('resize', resize);

            // Simulated world nodes
            const targetNode = { x: canvas.width * 0.75, y: canvas.height * 0.45 }; // WAF location
            const threatOrigins = [
                { x: canvas.width * 0.22, y: canvas.height * 0.40, name: 'North America' },
                { x: canvas.width * 0.50, y: canvas.height * 0.35, name: 'Europe' },
                { x: canvas.width * 0.80, y: canvas.height * 0.55, name: 'Asia East' },
                { x: canvas.width * 0.30, y: canvas.height * 0.70, name: 'South America' },
                { x: canvas.width * 0.88, y: canvas.height * 0.75, name: 'Australia' }
            ];

            let arcs = [];

            function spawnArc() {
                const origin = threatOrigins[Math.floor(Math.random() * threatOrigins.length)];
                arcs.push({
                    start: { x: origin.x, y: origin.y },
                    end: { x: targetNode.x, y: targetNode.y },
                    progress: 0,
                    speed: 0.015 + Math.random() * 0.02,
                    color: Math.random() > 0.4 ? 'rgba(255, 51, 102, 0.9)' : 'rgba(0, 242, 254, 0.9)'
                });
            }

            setInterval(spawnArc, 1200);

            function draw() {
                ctx.clearRect(0, 0, canvas.width, canvas.height);

                // Draw background grid lines
                ctx.strokeStyle = 'rgba(255, 255, 255, 0.03)';
                ctx.lineWidth = 1;
                for (let x = 0; x < canvas.width; x += 40) {
                    ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, canvas.height); ctx.stroke();
                }
                for (let y = 0; y < canvas.height; y += 40) {
                    ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(canvas.width, y); ctx.stroke();
                }

                // Draw origin nodes
                threatOrigins.forEach(node => {
                    ctx.beginPath();
                    ctx.arc(node.x, node.y, 4, 0, Math.PI * 2);
                    ctx.fillStyle = 'rgba(255, 51, 102, 0.6)';
                    ctx.fill();
                });

                // Draw WAF Gateway Destination Target
                ctx.beginPath();
                ctx.arc(targetNode.x, targetNode.y, 7, 0, Math.PI * 2);
                ctx.fillStyle = '#00f2fe';
                ctx.shadowColor = '#00f2fe';
                ctx.shadowBlur = 15;
                ctx.fill();
                ctx.shadowBlur = 0;

                // Draw Parabolic Attack Arcs
                for (let i = arcs.length - 1; i >= 0; i--) {
                    const arc = arcs[i];
                    arc.progress += arc.speed;

                    const curX = arc.start.x + (arc.end.x - arc.start.x) * arc.progress;
                    const midY = Math.min(arc.start.y, arc.end.y) - 50;
                    const curY = (1 - arc.progress) * (1 - arc.progress) * arc.start.y + 2 * (1 - arc.progress) * arc.progress * midY + arc.progress * arc.progress * arc.end.y;

                    // Glow particle
                    ctx.beginPath();
                    ctx.arc(curX, curY, 3.5, 0, Math.PI * 2);
                    ctx.fillStyle = arc.color;
                    ctx.shadowColor = arc.color;
                    ctx.shadowBlur = 10;
                    ctx.fill();
                    ctx.shadowBlur = 0;

                    if (arc.progress >= 1) {
                        arcs.splice(i, 1);
                    }
                }

                requestAnimationFrame(draw);
            }
            draw();
        }

        // Live Telemetry Poller
        async function pollTelemetry() {
            try {
                const res = await fetch('/api/stats');
                const stats = await res.json();

                document.getElementById('stat-total-requests').textContent = stats.total_requests;
                document.getElementById('stat-blocked-attacks').textContent = stats.blocked_attacks;
                document.getElementById('stat-challenged-requests').textContent = stats.challenged_requests || 0;
                document.getElementById('stat-latency').textContent = (stats.latency_ms || 0.38) + ' ms';

                const pct = stats.total_requests > 0 ? ((stats.blocked_attacks / stats.total_requests) * 100).toFixed(1) : '0';
                document.getElementById('stat-block-pct').textContent = pct + '%';

                // Update charts
                if (trafficChartInstance) {
                    const cleanData = trafficChartInstance.data.datasets[0].data;
                    const blockData = trafficChartInstance.data.datasets[1].data;

                    cleanData.shift();
                    cleanData.push(Math.max(0, stats.total_requests - stats.blocked_attacks));

                    blockData.shift();
                    blockData.push(stats.blocked_attacks);

                    trafficChartInstance.update('none');
                }

                // Update live incident stream
                fetchRecentEvents();
            } catch (err) { console.error(err); }
        }

        async function fetchRecentEvents() {
            try {
                const res = await fetch('/api/events');
                const events = await res.json();
                const feed = document.getElementById('live-incident-feed');
                if (!feed || !events || events.length === 0) return;

                feed.innerHTML = events.slice(0, 5).map(e => `
                    

                        

                            ${e.action}
                            ${e.client_ip}
                            ${e.path}
                        

                        ${e.timestamp.split(' ')[1] || e.timestamp}
                    

                `).join('');
            } catch (err) { console.error(err); }
        }

        // Audit Logs Table
        async function fetchAuditLogs() {
            const query = document.getElementById('log-search-query').value;
            const threat = document.getElementById('log-threat-type').value;

            try {
                const res = await fetch(`/api/logs?q=${encodeURIComponent(query)}&threat=${encodeURIComponent(threat)}`);
                const logs = await res.json();
                const tbody = document.getElementById('audit-log-body');

                if (!logs || logs.length === 0) {
                    tbody.innerHTML = `No matching log entries found`;
                    return;
                }

                tbody.innerHTML = logs.map(l => {
                    const encoded = encodeURIComponent(JSON.stringify(l));
                    let badgeClass = l.blocked ? 'badge-red' : (l.action.includes('Challenged') ? 'badge-amber' : 'badge-green');
                    return `
                        
                            ${l.timestamp}
                            ${l.client_ip}
                            ${l.path}
                            ${l.rule_id || 'N/A'}
                            ${l.action}
                            
                                
                            
                        
                    `;
                }).join('');
            } catch (err) { console.error(err); }
        }

        function inspectPayload(encodedJson) {
            try {
                const data = JSON.parse(decodeURIComponent(encodedJson));
                document.getElementById('modal-time').textContent = data.timestamp;
                document.getElementById('modal-ip').textContent = data.client_ip;
                document.getElementById('modal-path').textContent = data.path;
                document.getElementById('modal-rule').textContent = data.rule_id || 'N/A';
                document.getElementById('modal-reason').textContent = data.reason;
                document.getElementById('modal-payload').textContent = data.payload_snippet || 'No payload captured';
                openModal('inspectModal');
            } catch (e) { console.error(e); }
        }

        async function toggleSite(siteId, isChecked) {
            try {
                await fetch('/api/sites', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ site_id: siteId, defense_mode: isChecked })
                });
                showToast(`Site ID ${siteId} defense: ${isChecked ? 'Active' : 'Bypass'}`);
            } catch (err) { console.error(err); }
        }

        async function deleteSite(siteId) {
            if (!confirm('Delete protected site proxy rule?')) return;
            try {
                await fetch(`/api/sites/${siteId}`, { method: 'DELETE' });
                fetchSites();
                showToast('Site entry deleted');
            } catch (err) { console.error(err); }
        }

        async function submitNewSite() {
            const domain = document.getElementById('new-site-domain').value.trim();
            const upstream = document.getElementById('new-site-upstream').value.trim();
            if (!domain || !upstream) { alert('Enter domain and upstream'); return; }

            try {
                await fetch('/api/sites', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ domain, upstream })
                });
                closeModal('addSiteModal');
                fetchSites();
                showToast(`Protected domain added: ${domain}`);
            } catch (err) { console.error(err); }
        }

        // Terminal Console Helpers
        function appendTerminalLog(msg) {
            const term = document.getElementById('terminal');
            if (term) {
                const line = document.createElement('div');
                line.className = 'terminal-line';
                line.innerHTML = `[${new Date().toLocaleTimeString()}] ${msg}`;
                term.appendChild(line);
                term.scrollTop = term.scrollHeight;
            }
        }
        function clearTerminal() {
            const term = document.getElementById('terminal');
            if (term) term.innerHTML = '';
        }
